# The secrets that scripts/vault/Generate-Secrets.ps1 creates in Vault when they do not exist yet, one key per line:
# <mount> <secret> <key> <length>   a random value of that many characters (A-Z, a-z and 0-9, each kind at least once)
# <mount> <secret> <key> =<mount>/<secret>#<key>   the same value as another key, copied when this key is missing
#                                                  (the source is listed before it, so it exists by then)
# A key that exists is never changed. A new app adds its lines here and runs the script again.
# Only random values belong in this list. Values that must match reality are typed (Hetzner and NetBird token, Discord secret
# and webhooks, GitHub App key), database logins come from db_create.sh, S3 keys from setup-rustfs.sh and the values of the NetBird
# installer (NETBIRD_*, NB_PROXY_*) from the installer.

# --- argocd: the secrets of the apps in the cluster ----------------------------------------------------------------------------
argocd  harbor            admin.password           32
argocd  harbor            oidc.client_secret       32
argocd  jenkins           password                 32
argocd  jenkins           oidc.client_secret       32
argocd  prometheus        password                 32
argocd  prometheus        oidc.client_secret       32
argocd  vault             oidc.client_secret       32
argocd  argocd            oidc.client_secret       32
argocd  forgejo           admin.password           32
argocd  forgejo           oidc.client_secret       32
# security.secret_key encrypts the secrets in the Forgejo database: a new value makes them unreadable
argocd  forgejo           security.secret_key      64
argocd  forgejo           security.internal_token  64
# Restic password of the repository forgejo/restic (the CronJob forgejo-backup): without it the backup cannot be read
argocd  forgejo           backup.restic_password   64
argocd  developer-portal  oidc.client_secret       32
# Signs the service-to-service tokens inside the Developer Portal
argocd  developer-portal  backend.secret           64
# Encrypts saved Metabase connection credentials; preserve across restores/upgrades.
argocd  metabase          encryption.secret_key    64
